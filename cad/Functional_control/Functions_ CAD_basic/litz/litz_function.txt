from pathlib import Path
import sys

SCRIPT_PATH = Path(__file__).resolve()
CAD_DIR = SCRIPT_PATH.parents[3]
SYSTEM_DIR = CAD_DIR / "system"
SCRIPTS_DIR = CAD_DIR / "scripts"
for extra in (str(SYSTEM_DIR), str(SCRIPTS_DIR)):
    if extra not in sys.path:
        sys.path.insert(0, extra)

from CAD_basic import (  # type: ignore
    jingchengshu_wenjian,
    close_all_cad_processes,
    close_oldest_cad_process,
    start_applicationV9,
    li,
    last_obj,
    safe_delete,
)
from CAD_coordination import wait_quiescent  # type: ignore
from CAD_file_operations import draw_tarch_wall  # type: ignore


def litz(wait_timeout=60.0, retry_interval=1.5):
    """
    Diagnostic helper that ensures the active CAD session is TArch-enabled.

    Returns:
        bool: True if the probe succeeded, False otherwise.
    """
    import time

    def _tarch_probe(skip_li=False, allow_spawn=True):
        test_wall = None
        try:
            if not skip_li:
                if not li(allow_spawn=allow_spawn):
                    return False

            prev_handle = None
            try:
                prev = last_obj()
                prev_handle = getattr(prev, "Handle", None)
            except Exception:
                pass

            ok = draw_tarch_wall((0, 0, 0), (1000, 0, 0), thickness=240)
            if not ok:
                raise RuntimeError("draw_tarch_wall returned False")

            test_wall = last_obj()
            handle = getattr(test_wall, "Handle", None)
            object_name = getattr(test_wall, "ObjectName", "") or ""
            if not handle or handle == prev_handle:
                raise RuntimeError("failed to capture new wall handle")
            if "Wall" not in object_name and "TDb" not in object_name:
                raise RuntimeError(f"probe object is not a wall: {object_name}")

            try:
                safe_delete(test_wall)
            except Exception:
                try:
                    test_wall.Delete()
                except Exception:
                    pass
            print(f"[litz] wall probe OK handle={handle}")
            return True
        except Exception as exc:
            print(f"[litz] wall probe failed: {exc}")
            if test_wall is not None:
                try:
                    safe_delete(test_wall)
                except Exception:
                    try:
                        test_wall.Delete()
                    except Exception:
                        pass
            return False

    if _tarch_probe(skip_li=False, allow_spawn=True):
        print("[litz] current session already in TArch mode.")
        return True

    print("[litz] resetting CAD session by closing all processes.")
    try:
        close_all_cad_processes()
    except Exception as exc:
        print(f"[litz] close_all_cad_processes warning: {exc}")

    try:
        start_applicationV9(PTH=r"C:\Tangent\TArchT20V9", max_retries=3, retry_delay=2.0)
    except Exception as exc:
        print(f"[litz] failed to launch TArch: {exc}")
        return False

    wait_quiescent(min_quiet=2.0, timeout=45.0)

    def _normalize_processes(max_wait=30.0):
        deadline = time.time() + max_wait
        normalized = False
        while time.time() < deadline:
            cnt = jingchengshu_wenjian()
            if cnt == 1:
                normalized = True
                break
            if cnt == 0:
                time.sleep(0.5)
                continue
            print(f"[litz] detected {cnt} CAD processes, closing the oldest one.")
            try:
                close_oldest_cad_process()
            except Exception as exc:
                print(f"[litz] close_oldest_cad_process warning: {exc}")
            time.sleep(1.0)
        if not normalized:
            print("[litz] unable to reduce to a single CAD process within time limit.")

    _normalize_processes()

    try:
        global acad, doc, mp, sp  # type: ignore
        acad = doc = mp = sp = None  # type: ignore
    except Exception:
        pass

    deadline = time.time() + wait_timeout
    while time.time() < deadline:
        proc_cnt = jingchengshu_wenjian()
        if proc_cnt == 0:
            time.sleep(retry_interval)
            continue

        if _tarch_probe(skip_li=False, allow_spawn=False):
            print("[litz] TArch is ready after restart.")
            return True
        time.sleep(retry_interval)

    print("[litz] timeout waiting for TArch readiness.")
    return False
