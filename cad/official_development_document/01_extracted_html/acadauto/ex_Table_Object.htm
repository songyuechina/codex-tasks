Table Example

Using Programming Languages other than VBA

Sub Example_Table()
     ' This example adds a table in model space

    Dim MyModelSpace As AcadModelSpace
    Set MyModelSpace = ThisDrawing.ModelSpace
    Dim pt(2) As Double
    Dim MyTable As AcadTable
    
    ' Creates the table with an arbitrary number of cells
    Set MyTable = MyModelSpace.AddTable(pt, 5, 5, 10, 30)
    
    MsgBox "The current number of columns is: " & MyTable.Columns
    MsgBox "The current number of rows is: " & MyTable.Rows
    
    ' Delete 2 columns and 2 rows
    Call MyTable.DeleteColumns(1, 3)
    Call MyTable.DeleteRows(1, 3)
    
    MsgBox "The current number of columns is: " & MyTable.Columns
    MsgBox "The current number of rows is: " & MyTable.Rows
    
    ZoomExtents
    
End Sub
